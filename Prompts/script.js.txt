// Splash
setTimeout(() => {
    document.getElementById("splash").style.display = "none";
    document.getElementById("app").style.display = "block";
}, 2500);

// Navegação
function trocarTela(id){

    document.querySelectorAll(".screen").forEach(screen=>{
        screen.classList.remove("active");
    });

    document.getElementById(id).classList.add("active");
}

// Agenda
function adicionarCompromisso(){

    const data = document.getElementById("dataConsulta").value;
    const desc = document.getElementById("descricaoConsulta").value;

    if(data && desc){

        const li = document.createElement("li");

        li.innerHTML = `
            ${data} - ${desc}
            <button onclick="this.parentElement.remove()">Excluir</button>
        `;

        document.getElementById("listaCompromissos").appendChild(li);

    }
}

// Uploads
function uploadArquivo(event, pasta){

    const arquivo = event.target.files[0];

    if(!arquivo) return;

    const div = document.createElement("div");
    div.classList.add("arquivo");

    const link = document.createElement("a");

    link.href = URL.createObjectURL(arquivo);
    link.download = arquivo.name;
    link.innerText = arquivo.name;

    const excluir = document.createElement("button");
    excluir.innerText = "Excluir";

    excluir.onclick = ()=> div.remove();

    div.appendChild(link);
    div.appendChild(excluir);

    document.getElementById(pasta).appendChild(div);
}

// Medicamentos
function adicionarMedicamento(){

    const nome = document.getElementById("medicamento").value;
    const dosagem = document.getElementById("dosagem").value;
    const horario = document.getElementById("horario").value;

    if(nome){

        const tr = document.createElement("tr");

        tr.innerHTML = `
            <td>${nome}</td>
            <td>${dosagem}</td>
            <td>${horario}</td>
            <td>
                <button onclick="this.parentElement.parentElement.remove()">
                    Excluir
                </button>
            </td>
        `;

        document.getElementById("listaMedicamentos").appendChild(tr);

    }
}

// Busca
document.getElementById("search").addEventListener("keyup", function(){

    const termo = this.value.toLowerCase();

    document.querySelectorAll(".screen").forEach(screen => {

        if(screen.innerText.toLowerCase().includes(termo)){
            screen.style.display = "block";
        }else{
            screen.style.display = "none";
        }

    });

});
